import java.util.Scanner;

class Pet {
    String petName, breed;
    int age;

    void addPet(Scanner sc) {
        System.out.print("Enter Pet Name: ");
        petName = sc.nextLine();
        System.out.print("Enter Breed: ");
        breed = sc.nextLine();
        System.out.print("Enter Age: ");
        age = sc.nextInt();
        sc.nextLine();
    }

    void displayPet() {
        System.out.println("Pet Name: " + petName);
        System.out.println("Breed: " + breed);
        System.out.println("Age: " + age);
    }
}

class Owner {
    String name, phone;

    void addOwner(Scanner sc) {
        System.out.print("Enter Owner Name: ");
        name = sc.nextLine();
        System.out.print("Enter Phone: ");
        phone = sc.nextLine();
    }

    void displayOwner() {
        System.out.println("Owner Name: " + name);
        System.out.println("Phone: " + phone);
    }
}

class Vaccination {
    String vaccine, date;

    void addVaccination(Scanner sc) {
        System.out.print("Enter Vaccine Name: ");
        vaccine = sc.nextLine();
        System.out.print("Enter Vaccination Date: ");
        date = sc.nextLine();
    }

    void displayVaccination() {
        System.out.println("Vaccine: " + vaccine);
        System.out.println("Date: " + date);
    }
}

class Appointment {
    String date, time;

    void bookAppointment(Scanner sc) {
        System.out.print("Enter Appointment Date: ");
        date = sc.nextLine();
        System.out.print("Enter Appointment Time: ");
        time = sc.nextLine();
    }

    void displayAppointment() {
        System.out.println("Appointment Date: " + date);
        System.out.println("Appointment Time: " + time);
    }
}

public class PetCareManagementSystem {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        Pet pet = new Pet();
        Owner owner = new Owner();
        Vaccination vaccination = new Vaccination();
        Appointment appointment = new Appointment();

        int choice;

        do {
            System.out.println("\n===== PET CARE MANAGEMENT SYSTEM =====");
            System.out.println("1. Add Pet Details");
            System.out.println("2. Add Owner Details");
            System.out.println("3. Add Vaccination Details");
            System.out.println("4. Book Appointment");
            System.out.println("5. View Pet Records");
            System.out.println("6. Exit");
            System.out.print("Enter your choice: ");
            choice = sc.nextInt();
            sc.nextLine();

            switch (choice) {
                case 1:
                    pet.addPet(sc);
                    break;

                case 2:
                    owner.addOwner(sc);
                    break;

                case 3:
                    vaccination.addVaccination(sc);
                    break;

                case 4:
                    appointment.bookAppointment(sc);
                    break;

                case 5:
                    System.out.println("\n===== PET RECORD =====");
                    pet.displayPet();
                    owner.displayOwner();
                    vaccination.displayVaccination();
                    appointment.displayAppointment();
                    break;

                case 6:
                    System.out.println("Thank you!");
                    break;

                default:
                    System.out.println("Invalid choice!");
            }
        } while (choice != 6);

        sc.close();
    }
}
